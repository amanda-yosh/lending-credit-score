## Query

```sql
WITH janelas AS (
  SELECT COALESCE(on_behalf_of, borrower) AS wallet, project,
         MIN(block_time) AS ini, MAX(block_time) AS fim
  FROM lending.borrow
  WHERE project IN ('aave','compound')
    AND transaction_type = 'borrow'
  GROUP BY 1, 2
),
pares AS (
  SELECT a.wallet, a.project AS p1, b.project AS p2,
         GREATEST(a.ini, b.ini) AS overlap_ini,
         LEAST(a.fim, b.fim)    AS overlap_fim
  FROM janelas a JOIN janelas b
    ON a.wallet = b.wallet AND a.project < b.project
)
SELECT p1, p2,
       COUNT(*)                                        AS wallets_no_par,
       COUNT_IF(overlap_ini <= overlap_fim)            AS wallets_com_sobreposicao_temporal
FROM pares GROUP BY 1, 2 ORDER BY wallets_no_par DESC
```
