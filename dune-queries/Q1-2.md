## Query

```sql
-- Agregado (todas as chains). Usar COALESCE p/ testar a coluna de identidade decidida em Q0.1.
WITH b AS (
  SELECT COALESCE(on_behalf_of, borrower) AS wallet,
         ARRAY_AGG(DISTINCT project) AS projs
  FROM lending.borrow
  WHERE project IN ('aave','compound')
    AND transaction_type = 'borrow'
  GROUP BY 1
)
SELECT ARRAY_JOIN(ARRAY_SORT(projs), '+') AS combo, COUNT(*) AS wallets
FROM b GROUP BY 1 ORDER BY wallets DESC
```

```sql
-- Variante por chain (mesma wallet no mesmo protocolo, dentro de cada chain).
WITH b AS (
  SELECT blockchain, COALESCE(on_behalf_of, borrower) AS wallet,
         ARRAY_AGG(DISTINCT project) AS projs
  FROM lending.borrow
  WHERE project IN ('aave','compound')
    AND transaction_type = 'borrow'
  GROUP BY 1, 2
)
SELECT blockchain, ARRAY_JOIN(ARRAY_SORT(projs), '+') AS combo, COUNT(*) AS wallets
FROM b GROUP BY 1, 2 ORDER BY blockchain, wallets DESC
```
